\chapter{Implementation}\label{ch:impl}

\section{Project Resolution}
The compiler scans \verb|\input|, \verb|\include|, \verb|\includegraphics| and
\verb|\bibliography| commands and fetches each referenced file once.

\begin{definition}
A project is \emph{resolved} if every referenced file exists.
\end{definition}

\section{Complexity}

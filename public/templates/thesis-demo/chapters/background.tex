\chapter{Background}\label{ch:background}

\begin{definition}
A \emph{norm} on $\R^n$ is a map $\norm{\cdot}:\R^n\to\R_{\ge 0}$ with
$\norm{x+y}\le\norm{x}+\norm{y}$ and $\norm{\lambda x}=|\lambda|\norm{x}$.
\end{definition}

\begin{lemma}\label{lem:sum}
For all $n\in\N$: $\sum_{k=1}^{n} k = \frac{n(n+1)}{2}$.
\end{lemma}
\begin{proof}
By induction on $n$.
\end{proof}
